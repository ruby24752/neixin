<script>
    let score = {A:0,B:0,C:0};
    const resultData = {
        A:{
            title:"安稳守心型",
            text:`你偏爱平稳有序的生活，懂得劳逸结合。做事踏实，尽力而为，不盲目追逐。
<br/><br/>优势：情绪稳定，待人温和，擅长守住生活里的小美好。
<br/><br/>小提醒：偶尔可以大胆跳出固定节奏，尝试一点新的可能性。`
        },
        B:{
            title:"务实成长型",
            text:`目标感强，懂得灵活调整预期，不会被挫折困住。看重收获，遇事理性，聚焦事情本身。
<br/><br/>优势：抗压能力强，遇到问题愿意主动想办法解决。
<br/><br/>小提醒：不必事事追求结果，允许自己单纯享受过程。`
        },
        C:{
            title:"自由探索型",
            text:`你的内心不喜欢被束缚，优先守护自己和家人。想法灵活，有自己独立判断，不容易被别人影响。
<br/><br/>优势：拥有清晰边界感，忠于内心，懂得取舍。
<br/><br/>小提醒：有时候适当妥协，会减少很多不必要的内耗。`
        }
    };

    function select(el, nextPage){
        const val = el.getAttribute("data-val");
        score[val]++;
        //切换页面
        document.querySelectorAll('.page').forEach(p=>p.classList.remove('active'));
        document.getElementById(nextPage).classList.add('active');
        if(nextPage === "page-result"){
            showResult();
        }
    }

    function showResult(){
        let maxKey = "A";
        let maxNum = score.A;
        if(score.B>maxNum){maxKey="B";maxNum=score.B;}
        if(score.C>maxNum){maxKey="C";maxNum=score.C;}
        const res = resultData[maxKey];
        document.getElementById("result-title").innerText = res.title;
        document.getElementById("result-text").innerHTML = res.text;
    }

    function goPage(id){
        document.querySelectorAll('.page').forEach(p=>p.classList.remove('active'));
        document.getElementById(id).classList.add('active');
    }

    function restartTest(){
        score={A:0,B:0,C:0};
        goPage("page-cover");
    }
</script>
